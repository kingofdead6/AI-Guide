def train_softmax(X, y, K, lr=0.1, epochs=300, lam=1e-3):
    n, d = X.shape
    W = np.zeros((d, K)); Y = np.eye(K)[y]                 # one-hot (n, K)
    for _ in range(epochs):
        Z = X @ W
        Z -= Z.max(1, keepdims=True)
        P = np.exp(Z); P /= P.sum(1, keepdims=True)
        grad = X.T @ (P - Y) / n + lam * W                  # (d, K)
        W -= lr * grad
    return W

from sklearn.datasets import load_iris
X, y = load_iris(return_X_y=True)
X = np.c_[np.ones(len(X)), (X - X.mean(0)) / X.std(0)]
W = train_softmax(X, y, 3)
print("accuracy:", ((X @ W).argmax(1) == y).mean())
Practice problem
  1. Why is accuracy alone a bad training objective?
    Solution
    Accuracy is piecewise constant in w\mathbf ww: its gradient is zero almost everywhere, so gradient descent can't use it. Cross-entropy is a smooth, convex surrogate that also rewards calibrated confidence.
