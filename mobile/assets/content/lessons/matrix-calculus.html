Always check your gradients numerically

def num_grad(f, x, eps=1e-6):
    g = np.zeros_like(x)
    for i in range(x.size):
        e = np.zeros_like(x); e.flat[i] = eps
        g.flat[i] = (f(x + e) - f(x - e)) / (2 * eps)   # central difference
    return g

A = np.random.randn(3, 3); x = np.random.randn(3)
f = lambda v: v @ A @ v
print(np.allclose(num_grad(f, x), (A + A.T) @ x, atol=1e-5))   # True
