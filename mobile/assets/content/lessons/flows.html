Token-based generation

A VQ-VAE compresses an image or audio clip into a grid of discrete codes from a learned codebook (each encoder vector is snapped to its nearest code; a commitment loss keeps encoder outputs close to codes). A Transformer then models those codes autoregressively, exactly like text. Neural audio codecs do the same for sound, which is how many speech and music generators work. Video models typically generate in a compressed spatio-temporal latent space with Transformer backbones (diffusion Transformers).

def flow_matching_step(v_model, x1, opt):          # x1: batch of data
    x0 = torch.randn_like(x1)
    t = torch.rand(x1.shape[0], *[1] * (x1.dim() - 1), device=x1.device)
    xt = (1 - t) * x0 + t * x1
    loss = F.mse_loss(v_model(xt, t.flatten()), x1 - x0)
    opt.zero_grad(); loss.backward(); opt.step(); return loss.item()

@torch.no_grad()
def sample(v_model, shape, steps=50, device="cuda"):
    x = torch.randn(shape, device=device)
    for k in range(steps):                         # Euler integration from t=0 to t=1
        t = torch.full((shape[0],), k / steps, device=device)
        x = x + v_model(x, t) / steps
    return x
